# 1 เรื่อง route post ของ สร้าง ticket_images

## แบบเดิม 
// resident แจ้งซ่อม
router.post("/:ticket_id/images", residentMiddleware, uploadTicketBefore, uploadTicketImages)

// admin เพิ่มรูปจบงาน (เพิ่มทีหลังได้เลย)
router.post("/:ticket_id/completion", adminMiddleware, uploadTicketAfter, uploadCompletionImages)

## แบบใหม่ 

// resident แจ้งซ่อม
router.post("/:ticket_id/images/before", residentMiddleware, uploadTicketBefore, uploadTicketImages)

// admin เพิ่มรูปจบงาน (เพิ่มทีหลังได้เลย)
router.post("/:ticket_id/images/after", adminMiddleware, uploadTicketAfter, uploadCompletionImages)
### หรือ
//  ฝั่งลูกบ้าน: ส่งรูปตอน "เปิดเคส/แจ้งปัญหา" (Issue / Problem)
router.post("/:ticket_id/issue-images", residentMiddleware, uploadTicketBefore, uploadTicketImages);

// ฝั่งแอดมิน: ส่งรูปตอน "ปิดเคส/ทำเสร็จสิ้น" (Completion / Resolution)
router.post("/:ticket_id/completion", adminMiddleware, uploadTicketAfter, uploadCompletionImages);
